responses = {1: "Number is 1", 2: "Number is 2"}

Python

match number:
    case 1 | 2:
        print(f"Number is {number}")
    case _:
        print("Number isn't 1 or 2")

Java (14+)

String message = switch (number) {
    case 1, 2 -< "Number is " + number;
    default -< "Number isn't 1 or 2";
};
System.out.println(message);

Java

Map<Integer, String> messages = Map.of(
    1, "Number is 1",
    2, "Number is 2"
);

System.out.println(messages.getOrDefault(number, "Number isn't 1 or 2"));
