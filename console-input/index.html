    char username[50];

    printf("Enter your username: ");
    fgets(username, sizeof(username), stdin);

    username[strcspn(username, "\n")] = '\0';

    printf("Your username is %s", username);

    return 0;
}

Rust

use std::io::{self, Write};

    print!("Enter your username: ");
    io::stdout().flush().unwrap();

    let mut username = String::new();
    io::stdin().read_line(&mut username).unwrap();
    
    let username = username.trim_end(); 

    print!("Your username is {}", username);
}

GO

package main

import (
	"bufio"
	"fmt"
	"os"
	"strings"
)

func main() {
	reader := bufio.NewReader(os.Stdin)

	fmt.Print("Enter your username: ")
	username, _ := reader.ReadString('\n')
	username = strings.TrimSpace(username)

	fmt.Print("Your username is " + username)
}

Assembly

section .data
    prompt db "Enter your username: ", 0
    prompt_len equ $ - prompt
    
    output_msg db "Your username is ", 0
    output_len equ $ - output_msg

section .bss
    username resb 64

section .text
    global _start

_start:
    mov rax, 1
    mov rdi, 1
    mov rsi, prompt
    mov rdx, prompt_len
    syscall

    mov rax, 0
    mov rdi, 0
    mov rsi, username
    mov rdx, 64
    syscall
    mov rbx, rax

    mov rax, 1
    mov rdi, 1
    mov rsi, output_msg
    mov rdx, output_len
    syscall

    mov rax, 1
    mov rdi, 1
    mov rsi, username
    mov rdx, rbx
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall
